#usuarios


create table users(
id_users int not null auto_increment primary key,
nome varchar(120) not null,
email varchar(120) unique not null,
senha_hash varchar(255) not null,
data_de_nascimento year unsigned not null,
tipo_diabetes enum('1', '2') not null,
min_meta_glicemica int unsigned not null,
max_meta_glicemica int unsigned not null,
medicos_fk int,

foreign key(medicos_fk) references medicos(id_medicos)
) default charset = utf8mb4 ,engine= InnoDB;


#refeicoes


create table refeicoes(
id_refeicoes int not null auto_increment primary key,
user_fk int not null,
titulo varchar(40) not null,
observacao text,
cg int unsigned not null,
data_horario datetime not null,


foreign key(user_fk) references users(id_users)
on DELETE cascade
)default charset = utf8mb4 ,engine= InnoDB;

#N:N Alimentos refeicoes


create table alimentos_refeicoes(
alimentos_fk int not null,
refeicoes_fk int not null,
porcao_gramas int unsigned not null,


foreign key(refeicoes_fk) references refeicoes(id_refeicoes) on delete cascade,
foreign key(alimentos_fk) references alimentos(id_alimentos),


primary key(refeicoes_fk, alimentos_fk)
)default charset = utf8mb4 ,engine= InnoDB;

#view

Create or replace view full_rel_ali_ref as
SELECT  
u.id_users,
r.id_refeicoes,
a.id_alimentos,
al.porcao_gramas,
a.nome,
a.ig,
a.carb_por_grama,
r.titulo,
r.observacao,
r.cg,
r.data_horario
 from refeicoes as r
inner join alimentos_refeicoes as al on r.id_refeicoes = al.refeicoes_fk
INNER join alimentos as a on al.alimentos_fk = a.id_alimentos
Inner join users as u on r.user_fk = u.id_users;



#Glicemia


create table glicemia(
id_glicemia int not null auto_increment primary key,
user_fk int not null,
data_horario_registro datetime not null,
nivel_glicose int unsigned not null,
contexto varchar(30) not null,
refeicoes_fk int,
observacao text,

foreign key (user_fk) references users(id_users),
foreign key(refeicoes_fk) references refeicoes(id_refeicoes) on delete cascade
)default charset = utf8mb4 ,engine= InnoDB;
