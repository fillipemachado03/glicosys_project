async function get_unidade() {
  try {
    const res = await fetch("../backend/get_unidades.php", { method: "POST" });
    if (!res.ok) {
      throw new Error(`Erro HTTP: ${res.status}`);
    }
    let u = await res.json();
    console.log(u);
    return u;
  } catch (error) {
    console.log(error);
  }
}

async function get_medicos() {
  try {
    const res = await fetch("../../backend/Medicos.php", { method: "POST" });
    if (!res.ok) {
      throw new Error(`Erro HTTP: ${res.status}`);
    }

    let medicos = await res.json();
    console.log(res);

    return medicos;
  } catch (erro) {
    console.log(erro);
  }
}

function show_options(med) {
  let op = "<option value='null'>nenhum</option>";
  med.forEach((m) => {
    op += "<option value= " + m.nome + ">" + m.nome + "</option>";
  });
  return op;
}
